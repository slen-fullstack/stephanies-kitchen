  - { qty: 1, unit: "cup", item: "all-purpose flour" }
  - { qty: 0.25, unit: "cup", item: "sugar" }
  - { qty: 1, unit: "tbsp", item: "baking powder" }
  - { qty: 0.5, unit: "tsp", item: "salt" }
  - { qty: 1, unit: "cup", item: "milk" }
  - { qty: 0.5, unit: "cup", item: "melted butter" }
  - { qty: 2, unit: "", item: "eggs" }
  - { qty: 3, unit: "tbsp", item: "butter (for the honey butter)" }
  - { qty: 2, unit: "tbsp", item: "honey" }
steps:
  - "Heat the oven to 400°F (200°C) and butter an 8×8-inch pan."
  - "Whisk the cornmeal, flour, sugar, baking powder and salt."
  - "Add the milk, melted butter and eggs and stir until just combined."
  - "Pour into the pan and bake 20–25 minutes, until golden and a toothpick comes out clean."
  - "Melt the 3 tbsp butter with the honey and brush it over the warm cornbread."
